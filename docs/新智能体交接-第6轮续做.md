## 9. 真实交互检查脚本范式（直接改着用，存为 tools/round6-interact.js）
```js
const { chromium, webkit } = require('playwright');
const fs = require('fs');
const BASE = 'http://127.0.0.1:8899';
const PAGES = ['index.html','xinsheng.html','jiaotong.html','sushe.html','junxun.html',
  'xiaoqu.html','rushi.html','xueye.html','shenghuo.html','jiangzhu.html','shetuan.html',
  'zuzhi.html','jingsai.html','tice.html'];

async function runKernel(makeBrowser) {
  const browser = await makeBrowser();
  const page = await browser.newPage({ viewport: { width: 390, height: 844 } });
  const report = [];
  page.on('console', m => { if (m.type() === 'error') report.push('console.error: ' + m.text()); });
  page.on('pageerror', e => report.push('pageerror: ' + e.message));
  for (const p of PAGES) {
    const errs = [];
    await page.goto(BASE + '/' + p, { waitUntil: 'networkidle' });
    const els = await page.$$('button, a, [role="button"], input[type="button"], input[type="submit"]');
    for (const el of els) {
      const box = await el.boundingBox();
      if (!box || box.width === 0) continue;
      const info = await page.evaluate(e => {
        const href = e.getAttribute && e.getAttribute('href');
        return { tag: e.tagName, href: href || null, cls: (e.className || '').toString().slice(0, 30) };
      }, el);
      try {
        if (info.href && info.href.startsWith('#')) {
          await el.click({ timeout: 2000 });
          await page.waitForTimeout(120);
          const opened = await page.evaluate(sel => {
            const t = document.querySelector(sel);
            return !!t && t.getClientRects().length > 0;
          }, info.href);
          if (!opened) errs.push('anchor no effect: ' + info.href);
        } else if (info.href && (info.href.endsWith('.html') || info.href.startsWith('http'))) {
          await el.click({ timeout: 2000 });
          await page.waitForTimeout(300);
          const body = await page.evaluate(() => document.body ? document.body.innerText.length : 0);
          const is404 = await page.evaluate(() =>
            /404|Not Found|页面不存在/i.test(document.body ? document.body.innerText : '') &&
            (document.body ? document.body.innerText.length : 0) < 200);
          if (body < 50) errs.push('blank after nav: ' + info.href);
          if (is404) errs.push('404 after nav: ' + info.href);
          await page.goBack().catch(() => {});
          await page.waitForTimeout(150);
        } else {
          await el.click({ timeout: 2000 });
          await page.waitForTimeout(120);
        }
      } catch (e) {
        errs.push('click-fail [' + info.tag + '/' + info.cls + '] -> ' + e.message.split('\n')[0]);
      }
    }
    if (errs.length) report.push('[' + p + '] ' + errs.join(' | '));
  }
  await browser.close();
  return report;
}

(async () => {
  const chr = await runKernel(() => chromium.launch());
  const wk = await runKernel(() => webkit.launch());
  const all = [...chr, ...wk];
  fs.writeFileSync('.audit/interaction-bugs.md',
    '# 交互 Bug 检查\n\n' + (all.length ? all.map(l => '- ' + l).join('\n') : '✅ 双内核均无交互 bug'));
  console.log(all.length ? ('发现 ' + all.length + ' 处:\n' + all.join('\n')) : '✅ 全站交互无 bug');
})();
```
